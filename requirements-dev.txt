# Dev / CI dependencies — used to run the test suite locally and in CI.
# NEVER shipped to production: the Lambda/ECS image installs
# processor/requirements.txt (runtime deps), and the scientific stack (pandas,
# matplotlib, ...) is in the image at /opt/plot-stack (processor/stack-requirements.txt,
# pinned; install that for the exact versions).
#
# Install:  pip install -r requirements-dev.txt

pytest

# render() imports these at test time. In production they come from the image's stack# (processor/stack-requirements.txt), so they are intentionally absent from processor/requirements.txt;
# we pin them here only so the tests can run on a clean dev machine / CI.
pandas
matplotlib

# EDF reader for edf_processed_timeseries. pyedflib reads EDF slices and
# is in the image's stack in production.
pyedflib

# NWB reader (readers/nwb_to_signal.py) reads the HDF5 datasets directly
# with h5py — no pynwb, whose import alone costs ~15 s on a cold Lambda.
# Also in the image's stack in production.
h5py

# ts_dsp DSP tools: scipy provides the Butterworth filter (highpass_filter).
# Also in the image's stack in production.
scipy
